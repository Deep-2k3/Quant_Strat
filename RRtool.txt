        private NinjaTrader.NinjaScript.DrawingTools.Line tp2Line;
        
        private double prevEntryP, prevStopP, prevTp1P, prevTp2P;
        private bool isTrackingInitialized = false;
        private bool prevIsLong = true; 

        private string cacheKey = "";
        private static Dictionary<string, double> EntryCache = new Dictionary<string, double>();
        private static Dictionary<string, double> StopCache = new Dictionary<string, double>();

        private static Dictionary<string, double> TP2Cache = new Dictionary<string, double>();

        private Button toggleButton;

        private bool isToolActive = true;

        protected override void OnStateChange()
        {
            if (State == State.SetDefaults)
            {
                Description = "TradingView Style Risk/Reward Tool";
                Name = "RRtool";
                Calculate = Calculate.OnPriceChange;
                IsOverlay = true;
                DisplayInDataBox = false;
                PaintPriceMarkers = false;

                // Parameters
                RiskAmount = 500;
                DefaultStopDistance = 20; 
                ShowTP2 = true;
                AreaOpacity = 25;
                
                // Box Colors
                RiskBoxColor = Brushes.Red;
                TP1BoxColor = Brushes.LimeGreen;
                TP2BoxColor = Brushes.DarkGreen;

                // Text Colors
                EntryTextColor = Brushes.White;
                SLTextColor = Brushes.DarkMagenta;
                TP1TextColor = Brushes.LimeGreen;
                TP2TextColor = Brushes.DarkGreen;

                // Text Sizes
                EntryTextSize = 12;
                SLTextSize = 10;
                TP1TextSize = 10;
                TP2TextSize = 10;

                // Alignments
                EntryTextAlign = TextAlignment.Center;
                SLTextAlign = TextAlignment.Center;
                TP1TextAlign = TextAlignment.Center;
                TP2TextAlign = TextAlignment.Center;
            }
            else if (State == State.Historical)
            {
                cacheKey = ChartControl != null ? ChartControl.GetHashCode().ToString() + "_" + Instrument.FullName : Instrument.FullName;
                RemoveDrawObjects();

                if (ChartControl != null)
                {
                    ChartControl.Dispatcher.InvokeAsync(() =>
                    {
                        CreateWPFControls();

                        if (updateTimer == null)
                        {
                            updateTimer = new DispatcherTimer();
                            updateTimer.Interval = TimeSpan.FromMilliseconds(40);
                            updateTimer.Tick += UpdateTimer_Tick;
                            updateTimer.Start();
                        }
                    });
                }
            }
            else if (State == State.Terminated)
            {
                if (ChartControl != null)
                {
                    ChartControl.Dispatcher.InvokeAsync(() =>
                    {
                        DisposeWPFControls();
                        
                        if (updateTimer != null)
                        {
                            updateTimer.Stop();
                            updateTimer = null;
                        }
                    });
                }
            }
        }

        #region WPF UI Button Setup
        private void CreateWPFControls()
        {
            if (buttonContainer != null) return;

            toggleButton = new Button()
            {
                Content = "ON",
                Background = Brushes.LimeGreen,
                Foreground = Brushes.White,
                FontWeight = FontWeights.Bold,
                FontSize = 13,
                BorderBrush = Brushes.Transparent, 
                BorderThickness = new Thickness(0),
                Padding = new Thickness(15, 6, 15, 6),
                Cursor = Cursors.Arrow,
                Focusable = false
            };

            toggleButton.Click += ToggleButton_Click;

            buttonContainer = new Border()
            {
                Background = new SolidColorBrush(Color.FromArgb(220, 50, 50, 50)),
                BorderBrush = new SolidColorBrush(Color.FromArgb(255, 90, 90, 90)),
                BorderThickness = new Thickness(1.5),
                CornerRadius = new CornerRadius(4),
                Padding = new Thickness(3),

                HorizontalAlignment = HorizontalAlignment.Left,
                VerticalAlignment = VerticalAlignment.Top,

            };

            System.Windows.Controls.Panel.SetZIndex(buttonContainer, 99999);

                System.Windows.Controls.Grid.SetRow(buttonContainer, System.Windows.Controls.Grid.GetRow(ChartPanel));

        private void DisposeWPFControls()
        {
            if (buttonContainer != null)
            {
                if (toggleButton != null)
                    toggleButton.Click -= ToggleButton_Click;
                    
                ChartControl.Children.Remove(buttonContainer);
                buttonContainer = null;
                toggleButton = null;
            }
        }

        private void ToggleButton_Click(object sender, RoutedEventArgs e)
        {
            isToolActive = !isToolActive;
            
            if (isToolActive)
            {
                toggleButton.Content = "ON";
                toggleButton.Background = Brushes.LimeGreen;
                
                EntryCache.Remove(cacheKey);
                StopCache.Remove(cacheKey);
                TP1Cache.Remove(cacheKey);
                TP2Cache.Remove(cacheKey);

                isTrackingInitialized = false; 
                EnsureLinesExist();
            }
            else
            {
                toggleButton.Content = "OFF";
                toggleButton.Background = Brushes.DimGray;
                RemoveDrawObjects();
                isTrackingInitialized = false;
            }

            ChartControl?.Dispatcher.InvokeAsync(() => { ChartControl.InvalidateVisual(); });
        }

        private void UpdateTimer_Tick(object sender, EventArgs e)
        {
            try 
            {
                if (!isToolActive || ChartControl == null) return;

                EnsureLinesExist();
                SyncLines();
                
                ChartControl.Dispatcher.InvokeAsync(() => { ChartControl.InvalidateVisual(); });
            }
            catch { }
        }

        private void EnsureLinesExist()
        {
            if (Bars == null || Bars.Count == 0) return;

            double currentPrice = Bars.GetClose(Bars.Count - 1);

            if (GetLine("RR_Entry") == null)
            {
                double priceToUse = EntryCache.ContainsKey(cacheKey) ? EntryCache[cacheKey] : currentPrice; 
                CreateDraggableLine("RR_Entry", priceToUse);
            }

            if (GetLine("RR_Stop") == null)
            {
                double priceToUse = StopCache.ContainsKey(cacheKey) ? StopCache[cacheKey] : (currentPrice - DefaultStopDistance);
                CreateDraggableLine("RR_Stop", priceToUse);
            }
            
            if (GetLine("RR_Target1") == null)
            {
                double defaultDiff = (EntryCache.ContainsKey(cacheKey) ? EntryCache[cacheKey] : currentPrice) - (StopCache.ContainsKey(cacheKey) ? StopCache[cacheKey] : (currentPrice - DefaultStopDistance));
                double priceToUse = TP1Cache.ContainsKey(cacheKey) ? TP1Cache[cacheKey] : (currentPrice + defaultDiff);
                CreateDraggableLine("RR_Target1", priceToUse);
            }

            if (ShowTP2 && GetLine("RR_Target2") == null)
            {
                double defaultDiff = (EntryCache.ContainsKey(cacheKey) ? EntryCache[cacheKey] : currentPrice) - (StopCache.ContainsKey(cacheKey) ? StopCache[cacheKey] : (currentPrice - DefaultStopDistance));
                double priceToUse = TP2Cache.ContainsKey(cacheKey) ? TP2Cache[cacheKey] : (currentPrice + (defaultDiff * 2));
                CreateDraggableLine("RR_Target2", priceToUse);
            }
        }

        private void CreateDraggableLine(string tag, double price)
        {
            if (Bars == null || Bars.Count == 0) return;
            
            int currBarIdx = Bars.Count - 1;
            int startIdx = Math.Max(0, currBarIdx - 10); 
            DateTime startTime = Bars.GetTime(startIdx);
            
            TimeSpan barDuration = TimeSpan.FromMinutes(1); 
            if (BarsPeriod.BarsPeriodType == BarsPeriodType.Minute) barDuration = TimeSpan.FromMinutes(BarsPeriod.Value);
            else if (BarsPeriod.BarsPeriodType == BarsPeriodType.Second) barDuration = TimeSpan.FromSeconds(BarsPeriod.Value);
            else if (BarsPeriod.BarsPeriodType == BarsPeriodType.Day) barDuration = TimeSpan.FromDays(BarsPeriod.Value);

            DateTime endTime = Bars.GetTime(currBarIdx).Add(TimeSpan.FromTicks(barDuration.Ticks * 15)); 
            
            var l = Draw.Line(this, tag, false, startTime, price, endTime, price, Brushes.Transparent, DashStyleHelper.Solid, 6);
            l.IsLocked = false; 
        }

        private void RemoveDrawObjects()
        {
            RemoveDrawObject("RR_Entry");
            RemoveDrawObject("RR_Stop");
            RemoveDrawObject("RR_Target1");
            RemoveDrawObject("RR_Target2");
        }

        private void SyncLines()
        {
            entryLine = GetLine("RR_Entry");
            stopLine = GetLine("RR_Stop");
            tp1Line = GetLine("RR_Target1");
            tp2Line = GetLine("RR_Target2");

            if (entryLine == null || stopLine == null || tp1Line == null) return;
            
            if (!ShowTP2 && tp2Line != null)
            {
                RemoveDrawObject("RR_Target2");
                tp2Line = null;
            }

            double epsilon = 0.000001; 

            if (!isTrackingInitialized)
            {
                prevEntryP = entryLine.StartAnchor.Price;
                prevStopP = stopLine.StartAnchor.Price;
                prevTp1P = tp1Line.StartAnchor.Price;
                if (tp2Line != null) prevTp2P = tp2Line.StartAnchor.Price;
                
                prevIsLong = prevStopP < prevEntryP;
                isTrackingInitialized = true;
            }

            double currentEntryP = entryLine.StartAnchor.Price;
            double currentStopP = stopLine.StartAnchor.Price;
            double currentTp1P = tp1Line.StartAnchor.Price;
            double currentTp2P = tp2Line != null ? tp2Line.StartAnchor.Price : 0;
            bool isCurrentlyLong = currentStopP < currentEntryP;

            NinjaTrader.NinjaScript.DrawingTools.Line masterLine = entryLine; 
            if (stopLine.IsSelected) masterLine = stopLine;
            else if (tp1Line.IsSelected) masterLine = tp1Line;
            else if (tp2Line != null && tp2Line.IsSelected) masterLine = tp2Line;

            DateTime targetStart = masterLine.StartAnchor.Time;
            DateTime targetEnd = masterLine.EndAnchor.Time;

            if (!entryLine.IsSelected) { entryLine.StartAnchor.Time = targetStart; entryLine.EndAnchor.Time = targetEnd; }
            if (!stopLine.IsSelected) { stopLine.StartAnchor.Time = targetStart; stopLine.EndAnchor.Time = targetEnd; }
            if (!tp1Line.IsSelected) { tp1Line.StartAnchor.Time = targetStart; tp1Line.EndAnchor.Time = targetEnd; }
            if (tp2Line != null && !tp2Line.IsSelected) { tp2Line.StartAnchor.Time = targetStart; tp2Line.EndAnchor.Time = targetEnd; }
            
            if (Math.Abs(currentEntryP - prevEntryP) > epsilon)
            {
                double change = currentEntryP - prevEntryP;
                currentStopP += change;
                currentTp1P += change;
                if (tp2Line != null) currentTp2P += change;

                if (!stopLine.IsSelected) { stopLine.StartAnchor.Price = currentStopP; stopLine.EndAnchor.Price = currentStopP; }
                if (!tp1Line.IsSelected) { tp1Line.StartAnchor.Price = currentTp1P; tp1Line.EndAnchor.Price = currentTp1P; }
                if (tp2Line != null && !tp2Line.IsSelected) { tp2Line.StartAnchor.Price = currentTp2P; tp2Line.EndAnchor.Price = currentTp2P; }
            }
            else if (Math.Abs(currentStopP - prevStopP) > epsilon)
            {
                double riskDist = Math.Abs(currentEntryP - currentStopP);
                if (riskDist < TickSize) riskDist = TickSize;

                if (isCurrentlyLong)
                {
                    currentTp1P = currentEntryP + riskDist;
                    if (tp2Line != null) currentTp2P = currentEntryP + (riskDist * 2);
                }
                else
                {
                    currentTp1P = currentEntryP - riskDist;
                    if (tp2Line != null) currentTp2P = currentEntryP - (riskDist * 2);
                }

                if (!tp1Line.IsSelected) { tp1Line.StartAnchor.Price = currentTp1P; tp1Line.EndAnchor.Price = currentTp1P; }
                if (tp2Line != null && !tp2Line.IsSelected) { tp2Line.StartAnchor.Price = currentTp2P; tp2Line.EndAnchor.Price = currentTp2P; }
                
                prevIsLong = isCurrentlyLong;
            }
            
            if (!entryLine.IsSelected && Math.Abs(entryLine.EndAnchor.Price - currentEntryP) > epsilon) entryLine.EndAnchor.Price = currentEntryP;
            if (!stopLine.IsSelected && Math.Abs(stopLine.EndAnchor.Price - currentStopP) > epsilon) stopLine.EndAnchor.Price = currentStopP;
            if (!tp1Line.IsSelected && Math.Abs(tp1Line.EndAnchor.Price - currentTp1P) > epsilon) tp1Line.EndAnchor.Price = currentTp1P;
            if (tp2Line != null && !tp2Line.IsSelected && Math.Abs(tp2Line.EndAnchor.Price - currentTp2P) > epsilon) tp2Line.EndAnchor.Price = currentTp2P;

            prevEntryP = currentEntryP;
            prevStopP = currentStopP;
            prevTp1P = currentTp1P;
            if (tp2Line != null) prevTp2P = currentTp2P;

            EntryCache[cacheKey] = prevEntryP;
            StopCache[cacheKey] = prevStopP;
            TP1Cache[cacheKey] = prevTp1P;
            if (tp2Line != null) TP2Cache[cacheKey] = prevTp2P;
        }

        private SharpDX.DirectWrite.TextAlignment GetDXAlignment(TextAlignment align)
        {
            if (align == TextAlignment.Center) return SharpDX.DirectWrite.TextAlignment.Center;
            if (align == TextAlignment.Right) return SharpDX.DirectWrite.TextAlignment.Trailing;
            return SharpDX.DirectWrite.TextAlignment.Leading;
        }

        protected override void OnRender(ChartControl chartControl, ChartScale chartScale)
        {
            base.OnRender(chartControl, chartScale);

            if (!isToolActive || Bars == null || ChartPanel == null) return;
            if (entryLine == null || stopLine == null || tp1Line == null) return;

            NinjaTrader.NinjaScript.DrawingTools.Line masterLine = entryLine; 
            if (stopLine.IsSelected) masterLine = stopLine;
            else if (tp1Line.IsSelected) masterLine = tp1Line;
            else if (tp2Line != null && tp2Line.IsSelected) masterLine = tp2Line;

            float startX = chartControl.GetXByTime(masterLine.StartAnchor.Time);
            float endX = chartControl.GetXByTime(masterLine.EndAnchor.Time);
            float width = endX - startX;
            if (width <= 0) width = 1;

            float eY = chartScale.GetYByValue(entryLine.StartAnchor.Price);
            float sY = chartScale.GetYByValue(stopLine.StartAnchor.Price);
            float t1Y = chartScale.GetYByValue(tp1Line.StartAnchor.Price);

            var scbRisk = RiskBoxColor as SolidColorBrush;
            var scbTp1 = TP1BoxColor as SolidColorBrush;
            var scbTp2 = TP2BoxColor as SolidColorBrush;
            
            var txtEntryC = EntryTextColor as SolidColorBrush;
            var txtSlC = SLTextColor as SolidColorBrush;
            var txtTp1C = TP1TextColor as SolidColorBrush;
            var txtTp2C = TP2TextColor as SolidColorBrush;

            if (scbRisk == null || scbTp1 == null || scbTp2 == null) return;

            using (var dxRisk = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)scbRisk.Color.R, (int)scbRisk.Color.G, (int)scbRisk.Color.B, (int)(AreaOpacity * 2.55f))))
            using (var dxTp1 = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)scbTp1.Color.R, (int)scbTp1.Color.G, (int)scbTp1.Color.B, (int)(AreaOpacity * 2.55f))))
            using (var dxTp2 = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)scbTp2.Color.R, (int)scbTp2.Color.G, (int)scbTp2.Color.B, (int)(AreaOpacity * 2.55f))))
            
            using (var brEntry = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)txtEntryC.Color.R, (int)txtEntryC.Color.G, (int)txtEntryC.Color.B, 255)))
            using (var brSl = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)txtSlC.Color.R, (int)txtSlC.Color.G, (int)txtSlC.Color.B, 255)))
            using (var brTp1 = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)txtTp1C.Color.R, (int)txtTp1C.Color.G, (int)txtTp1C.Color.B, 255)))
            using (var brTp2 = new SharpDX.Direct2D1.SolidColorBrush(RenderTarget, new SharpDX.Color((int)txtTp2C.Color.R, (int)txtTp2C.Color.G, (int)txtTp2C.Color.B, 255)))
            
            using (var fmtEntry = new SharpDX.DirectWrite.TextFormat(NinjaTrader.Core.Globals.DirectWriteFactory, "Arial", SharpDX.DirectWrite.FontWeight.Bold, SharpDX.DirectWrite.FontStyle.Normal, EntryTextSize))
            using (var fmtSl = new SharpDX.DirectWrite.TextFormat(NinjaTrader.Core.Globals.DirectWriteFactory, "Arial", SharpDX.DirectWrite.FontWeight.Bold, SharpDX.DirectWrite.FontStyle.Normal, SLTextSize))
            using (var fmtTp1 = new SharpDX.DirectWrite.TextFormat(NinjaTrader.Core.Globals.DirectWriteFactory, "Arial", SharpDX.DirectWrite.FontWeight.Bold, SharpDX.DirectWrite.FontStyle.Normal, TP1TextSize))
            using (var fmtTp2 = new SharpDX.DirectWrite.TextFormat(NinjaTrader.Core.Globals.DirectWriteFactory, "Arial", SharpDX.DirectWrite.FontWeight.Bold, SharpDX.DirectWrite.FontStyle.Normal, TP2TextSize))
            
            using (var strokeStyleDash = new SharpDX.Direct2D1.StrokeStyle(RenderTarget.Factory, new SharpDX.Direct2D1.StrokeStyleProperties { DashStyle = SharpDX.Direct2D1.DashStyle.Dash }))
            {
                fmtEntry.TextAlignment = GetDXAlignment(EntryTextAlign);
                fmtSl.TextAlignment = GetDXAlignment(SLTextAlign);
                fmtTp1.TextAlignment = GetDXAlignment(TP1TextAlign);
                fmtTp2.TextAlignment = GetDXAlignment(TP2TextAlign);

                // ZONES
                var riskRect = new SharpDX.RectangleF(startX, Math.Min(eY, sY), width, Math.Abs(eY - sY));
                RenderTarget.FillRectangle(riskRect, dxRisk);

                var rw1Rect = new SharpDX.RectangleF(startX, Math.Min(eY, t1Y), width, Math.Abs(eY - t1Y));
                RenderTarget.FillRectangle(rw1Rect, dxTp1);

                float t2Y = 0;
                if (ShowTP2 && tp2Line != null)
                {
                    t2Y = chartScale.GetYByValue(tp2Line.StartAnchor.Price);
                    var rw2Rect = new SharpDX.RectangleF(startX, Math.Min(t1Y, t2Y), width, Math.Abs(t1Y - t2Y));
                    RenderTarget.FillRectangle(rw2Rect, dxTp2);
                }

                // DRAW ENTRY/SL BORDER LINES (Removed TP Lines for cleaner look)
                RenderTarget.DrawLine(new SharpDX.Vector2(startX, eY), new SharpDX.Vector2(endX, eY), brEntry, 2, strokeStyleDash);

                // TEXT
                double currentEntryP = entryLine.StartAnchor.Price;
                double currentStopP = stopLine.StartAnchor.Price;
                double currentTp1P = tp1Line.StartAnchor.Price;
                
                double riskDist = Math.Abs(currentEntryP - currentStopP);
                if (riskDist < TickSize) riskDist = TickSize;

                double pointValue = Instrument.MasterInstrument.PointValue;
                double riskPerOne = riskDist * pointValue;
                if (riskPerOne <= 0) riskPerOne = 1; 

                double qty = RiskAmount / riskPerOne;
                if (qty <= 0) qty = 0.01;

                string mainLabel = string.Format("RISK: ${0}  |  QTY: {1:0.00}  |  DIST: {2:0.00} pts", RiskAmount, qty, riskDist);
                float midRiskY = (eY + sY) / 2;
                
                RenderTarget.DrawText(mainLabel, fmtEntry, new SharpDX.RectangleF(startX + 5, midRiskY - (EntryTextSize/2f), width - 10, EntryTextSize * 2), brEntry);
                RenderTarget.DrawText(currentStopP.ToString("N2"), fmtSl, new SharpDX.RectangleF(startX + 5, sY + 5, width - 10, SLTextSize * 2), brSl);

                double tp1Dist = Math.Abs(currentTp1P - currentEntryP);
                string t1Label = string.Format("Target 1: {0:N2} ({1:0.00} pts) | +${2:N2}", currentTp1P, tp1Dist, tp1Dist * pointValue * qty);
                RenderTarget.DrawText(t1Label, fmtTp1, new SharpDX.RectangleF(startX + 5, t1Y - (TP1TextSize + 5), width - 10, TP1TextSize * 2), brTp1);

                if (ShowTP2 && tp2Line != null)
                {
                    double currentTp2P = tp2Line.StartAnchor.Price;
                    double tp2Dist = Math.Abs(currentTp2P - currentEntryP);
                    string t2Label = string.Format("Target 2: {0:N2} ({1:0.00} pts) | +${2:N2}", currentTp2P, tp2Dist, tp2Dist * pointValue * qty);
                    RenderTarget.DrawText(t2Label, fmtTp2, new SharpDX.RectangleF(startX + 5, t2Y - (TP2TextSize + 5), width - 10, TP2TextSize * 2), brTp2);
                }
            }
        }

        private NinjaTrader.NinjaScript.DrawingTools.Line GetLine(string tag)
        {
            foreach (DrawingTool dt in DrawObjects)
            {
                if (dt.Tag == tag && dt is NinjaTrader.NinjaScript.DrawingTools.Line) 
                    return dt as NinjaTrader.NinjaScript.DrawingTools.Line;
            }
            return null;
        }

		public RRtool RRtool(ISeries<double> input, double riskAmount, double defaultStopDistance, bool showTP2, int areaOpacity, int entryTextSize, int sLTextSize, int tP1TextSize, int tP2TextSize, TextAlignment entryTextAlign, TextAlignment sLTextAlign, TextAlignment tP1TextAlign, TextAlignment tP2TextAlign)
		{
			if (cacheRRtool != null)
				for (int idx = 0; idx < cacheRRtool.Length; idx++)
					if (cacheRRtool[idx] != null && cacheRRtool[idx].RiskAmount == riskAmount && cacheRRtool[idx].DefaultStopDistance == defaultStopDistance && cacheRRtool[idx].ShowTP2 == showTP2 && cacheRRtool[idx].AreaOpacity == areaOpacity && cacheRRtool[idx].EntryTextSize == entryTextSize && cacheRRtool[idx].SLTextSize == sLTextSize && cacheRRtool[idx].TP1TextSize == tP1TextSize && cacheRRtool[idx].TP2TextSize == tP2TextSize && cacheRRtool[idx].EntryTextAlign == entryTextAlign && cacheRRtool[idx].SLTextAlign == sLTextAlign && cacheRRtool[idx].TP1TextAlign == tP1TextAlign && cacheRRtool[idx].TP2TextAlign == tP2TextAlign && cacheRRtool[idx].EqualsInput(input))
						return cacheRRtool[idx];
			return CacheIndicator<RRtool>(new RRtool(){ RiskAmount = riskAmount, DefaultStopDistance = defaultStopDistance, ShowTP2 = showTP2, AreaOpacity = areaOpacity, EntryTextSize = entryTextSize, SLTextSize = sLTextSize, TP1TextSize = tP1TextSize, TP2TextSize = tP2TextSize, EntryTextAlign = entryTextAlign, SLTextAlign = sLTextAlign, TP1TextAlign = tP1TextAlign, TP2TextAlign = tP2TextAlign }, input, ref cacheRRtool);
		}
	}
